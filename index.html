const typingArea = document.getElementById('typing-area'); const cursorEl = document.getElementById('cursor'); let gravity = 'down'; let elements = []; let cursor = {x: 10, y: 10}; const letterSize = 20; const appleSize = 30; const gravityOrder = ['down', 'left', 'up', 'right']; let lettersBuffer = []; // 각 입력 위치를 저장 function updateCursor() { cursorEl.style.left = (typingArea.offsetLeft + cursor.x) + 'px'; cursorEl.style.top = (typingArea.offsetTop + cursor.y) + 'px'; cursorEl.style.height = letterSize + 'px'; } typingArea.addEventListener('click', (e) => { const rect = typingArea.getBoundingClientRect(); cursor.x = e.clientX - rect.left; cursor.y = e.clientY - rect.top; updateCursor(); typingArea.focus(); lettersBuffer = []; // 새 입력 위치를 누르면 버퍼 리셋 }); function createLetter(char) { const el = document.createElement('span'); el.className = 'letter'; el.textContent = char; el.style.left = cursor.x + 'px'; el.style.top = cursor.y + 'px'; typingArea.appendChild(el); elements.push({el, type:'letter', x: cursor.x, y: cursor.y}); lettersBuffer.push({el, x: cursor.x, y: cursor.y}); // 위치 저장 moveElement(el, 'letter'); // 다음 입력 커서 위치(오른쪽으로 이동) cursor.x += letterSize; updateCursor(); } function createApple() { const el = document.createElement('img'); el.src = 'apple.png'; el.className = 'apple'; el.style.left = cursor.x + 'px'; el.style.top = cursor.y + 'px'; typingArea.appendChild(el); elements.push({el, type:'apple', x: cursor.x, y: cursor.y}); lettersBuffer.push({el, x: cursor.x, y: cursor.y}); // 사과도 버퍼에 추가 moveElement(el, 'apple'); cursor.x += appleSize; // 사과 크기만큼 커서 이동 updateCursor(); } function moveElement(el, type) { const area = typingArea.getBoundingClientRect(); let top = parseFloat(el.style.top); let left = parseFloat(el.style.left); const size = type === 'letter' ? letterSize : appleSize; function isBlocked(nextX, nextY) { for (let obj of elements) { if (obj.el === el) continue; const ox = parseFloat(obj.el.style.left); const oy = parseFloat(obj.el.style.top); const osize = obj.type === 'letter' ? letterSize : appleSize; if (nextX < ox + osize && nextX + size > ox && nextY < oy + osize && nextY + size > oy) { return true; } } return false; } function step() { let moved = false; if (gravity === 'down') { if (top + size + 2 <= area.height && !isBlocked(left, top+2)) { top += 2; moved = true; } else if (type === 'apple') rotateGravity(); } else if (gravity === 'left') { if (left - 2 >= 0 && !isBlocked(left-2, top)) { left -= 2; moved = true; } else if (type === 'apple') rotateGravity(); } else if (gravity === 'up') { if (top - 2 >= 0 && !isBlocked(left, top-2)) { top -= 2; moved = true; } else if (type === 'apple') rotateGravity(); } else if (gravity === 'right') { if (left + size + 2 <= area.width && !isBlocked(left+2, top)) { left += 2; moved = true; } else if (type === 'apple') rotateGravity(); } el.style.top = top + 'px'; el.style.left = left + 'px'; if (moved) requestAnimationFrame(step); } step(); } function rotateGravity() { gravity = gravityOrder[(gravityOrder.indexOf(gravity) + 1) % 4]; } typingArea.addEventListener('keydown', (e) => { if (e.key === 'Escape') { elements.forEach(o => o.el.remove()); elements = []; lettersBuffer = []; gravity = 'down'; cursor.x = 10; // 초기화 cursor.y = 10; updateCursor(); } else if (e.key === 'Enter') { const text = typingArea.textContent.trim(); if (!text) return; if (text.toLowerCase() === 'apple') { createApple(); } else { for (let char of text) createLetter(char); } typingArea.textContent = ''; } else if (e.key.length === 1) { typingArea.textContent += e.key; createLetter(e.key); } else if (e.key === 'Backspace') { if (lettersBuffer.length > 0) { const last = lettersBuffer.pop(); if (last.el) last.el.remove(); // 커서 위치 되돌리기 if (lettersBuffer.length > 0) { cursor.x = lettersBuffer[lettersBuffer.length-1].x + letterSize; cursor.y = lettersBuffer[lettersBuffer.length-1].y; } else { cursor.x = 10; cursor.y = 10; } updateCursor(); } // 실제 타이핑 영역 text도 삭제 typingArea.textContent = typingArea.textContent.slice(0,-1); } updateCursor(); }); updateCursor(); typingArea.focus();
